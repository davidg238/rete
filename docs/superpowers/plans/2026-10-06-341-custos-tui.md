# custos TUI Implementation Plan (#341)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `custos tui [--url URL]` — a Bubble Tea front over a serving custos's `/debug` and `/profile`, an HTTP client and nothing more.

**Architecture:** `custos/internal/tui` holds a `Client` (the five endpoints, typed with the server's own exported answer types) and one Bubble Tea `Model` with two tabs. A single long-poll on `/debug/state?after=Seq` feeds the model; commands post to `/debug/do`. The real-VM gate drives the model with key messages, no terminal, against a custos the existing Python gates start.

**Tech Stack:** Go 1.26 (custos module), `github.com/charmbracelet/bubbletea v1.3.10`, `github.com/charmbracelet/lipgloss v1.1.0`.

**Spec:** `docs/design/2026-10-06-custos-tui-design.md`

## Global Constraints

- The TUI talks only to `POST /debug/do`, `GET /debug/state?after=N`, `GET /debug/source`, `POST /profile/capture?ms=N`, `GET /profile/last` (D1). No `debug.Session`, no cable, no prose parsing.
- Colours: the terminal's 16 ANSI colours only, `lipgloss.Color("0")`–`("15")` (D3). No hex.
- `custos tui` is a subcommand of the custos binary; code in `custos/internal/tui` (D4).
- Nothing sleeps to wait for a stop; the only waits are the long-poll and the 1 s back-off after an HTTP error.
- No `bubbles` dependency (spec D4 amended by this plan: two hand-rolled widgets — a one-line input and a list — cost less than a third module).
- Every commit: explicit paths (never `-a`), message ends `refs #341`.

## Review Focus

1. custos not running / wrong `--url` → the TUI starts, says so in the status line, keeps retrying; never exits except on `q`. (Task 3 test `TestUnreachable`.)
2. custos serving profile only (debug routes 404) → the debug tab says "no debug target" and the profile tab still works. (Task 3 test `TestNoDebugTarget`.)
3. A key pressed while a command is in flight → refused locally with a status line, not queued (the server would 409). (Task 3 test `TestBusy`.)
4. A program that ends (`Run = ended`) or a lost cable (`Run = lost`) → shown in the status line; keys still work, nothing panics on a nil `Stop`. (Task 3 test `TestEndedNilStop`.)
5. A terminal smaller than the panes, or no `WindowSizeMsg` yet → `View()` renders without panicking (negative widths clamp). (Task 4 test `TestTinyTerminal`.)

```go
// Copyright (c) 2026 Ekorau LLC

package tui

// client.go — custos's HTTP surface as the TUI reads it (rete#341 D1): the
// endpoints the page and an agent use, typed with the server's own answers.

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"strings"
	"time"

	"github.com/davidg238/rete/custos/internal/custos"
	"github.com/davidg238/rete/custos/internal/debug"
)

// Client talks to one serving custos.
type Client struct {
	Base string
	HTTP *http.Client
}

// NewClient's timeout outlasts the server's 25 s long-poll and a long capture.
func NewClient(base string) *Client {
	return &Client{strings.TrimRight(base, "/"), &http.Client{Timeout: 70 * time.Second}}
}

// StatusError is a non-2xx answer: 404 (no such target configured), 409 (one
// already in flight), 502 (the capture's program ended early).
type StatusError struct {
	Code int
	Body string
}

func (e *StatusError) Error() string {
	return fmt.Sprintf("%d %s", e.Code, strings.TrimSpace(e.Body))
}

func (c *Client) call(method, path, body string, v any) error {
	req, err := http.NewRequest(method, c.Base+path, strings.NewReader(body))
	if err != nil {
		return err
	}
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return err
	}
	defer resp.Body.Close()
	if resp.StatusCode/100 != 2 {
		b, _ := io.ReadAll(io.LimitReader(resp.Body, 512))
		return &StatusError{resp.StatusCode, string(b)}
	}
	return json.NewDecoder(resp.Body).Decode(v)
}

func (c *Client) Source() (s custos.Source, err error) {
	return s, c.call("GET", "/debug/source", "", &s)
}

func (c *Client) State(after int) (s debug.State, err error) {
	return s, c.call("GET", fmt.Sprintf("/debug/state?after=%d", after), "", &s)
}

func (c *Client) Do(cmd string) (d custos.Done, err error) {
	return d, c.call("POST", "/debug/do", cmd, &d)
}

func (c *Client) Capture(ms int) (p custos.Capture, err error) {
	return p, c.call("POST", fmt.Sprintf("/profile/capture?ms=%d", ms), "", &p)
}

func (c *Client) Last() (p custos.Capture, err error) {
	return p, c.call("GET", "/profile/last", "", &p)
}
```

```go
// Copyright (c) 2026 Ekorau LLC

package tui

import (
	"testing"
	"time"

	tea "github.com/charmbracelet/bubbletea"
)

// drive runs cmd and every command the model answers with, until until(m)
// holds or timeout passes. A tea.BatchMsg is expanded; quit is ignored.
func drive(t *testing.T, m Model, cmd tea.Cmd, timeout time.Duration, until func(Model) bool) Model {
	t.Helper()
	msgs := make(chan tea.Msg, 64)
	run := func(c tea.Cmd) {
		if c != nil {
			go func() { msgs <- c() }()
		}
	}
	run(cmd)
	deadline := time.After(timeout)
	for !until(m) {
		select {
		case msg := <-msgs:
			switch msg := msg.(type) {
			case tea.BatchMsg:
				for _, c := range msg {
					run(c)
				}
				continue
			case tea.QuitMsg, nil:
				continue
			}
			nm, c := m.Update(msg)
			m = nm.(Model)
			run(c)
		case <-deadline:
			t.Fatalf("timed out; status=%q run=%q stop=%+v", m.status, m.st.Run, m.st.Stop)
		}
	}
	return m
}

// press delivers keys in order and answers the batch of commands they asked for.
func press(t *testing.T, m Model, keys ...string) (Model, tea.Cmd) {
	t.Helper()
	var cmds []tea.Cmd
	for _, k := range keys {
		var msg tea.KeyMsg
		switch k {
		case "enter":
			msg = tea.KeyMsg{Type: tea.KeyEnter}
		case "esc":
			msg = tea.KeyMsg{Type: tea.KeyEsc}
		case "backspace":
			msg = tea.KeyMsg{Type: tea.KeyBackspace}
		case "tab":
			msg = tea.KeyMsg{Type: tea.KeyTab}
		case "up":
			msg = tea.KeyMsg{Type: tea.KeyUp}
		case "down":
			msg = tea.KeyMsg{Type: tea.KeyDown}
		default:
			msg = tea.KeyMsg{Type: tea.KeyRunes, Runes: []rune(k)}
		}
		nm, c := m.Update(msg)
		m = nm.(Model)
		cmds = append(cmds, c)
	}
	return m, tea.Batch(cmds...)
}
```

```go
// Copyright (c) 2026 Ekorau LLC

// Package tui is `custos tui` (rete#341): a terminal front over a serving
// custos's /debug and /profile. It is an HTTP client like the page and an
// agent (design 2026-10-06-custos-tui-design.md D1); custos owns the cable.
package tui

import (
	"errors"
	"fmt"
	"path/filepath"
	"slices"
	"strconv"
	"time"

	tea "github.com/charmbracelet/bubbletea"

	"github.com/davidg238/rete/custos/internal/custos"
	"github.com/davidg238/rete/custos/internal/debug"
)

type tab int

const (
	debugTab tab = iota
	profileTab
)

// Model is the whole TUI. Exactly one long-poll is in flight at a time; it is
// re-issued only from the stateMsg it answers.
type Model struct {
	c        *Client
	tab      tab
	w, h     int
	src      custos.Source
	st       debug.State
	cursor   int    // the source line under the cursor, 1-based
	frame    int    // the frame whose locals are asked for
	prompt   string // non-empty while the one-line input is open: ":" or "window ms: "
	input    string
	status   string // the last refusal or error; "" when the last thing worked
	inflight string // the command posted and not yet answered
	btFor    string // the stop a bt was already sent for
	noDebug  bool   // /debug answered 404: custos serves no debug target
	prof     *custos.Capture
	ms       int // the last profile window
	capturing bool
}

type sourceMsg struct {
	src custos.Source
	err error
}
type stateMsg struct {
	st  debug.State
	err error
}
type doneMsg struct {
	cmd string
	d   custos.Done
	err error
}
type profMsg struct {
	c   custos.Capture
	err error
	last bool // from /profile/last at start-up: a 404 there is not an error
}

// New answers the model for c; Init starts the source fetch and the long-poll.
func New(c *Client) Model { return Model{c: c, cursor: 1, ms: 3000} }

func (m Model) Init() tea.Cmd {
	c := m.c
	return tea.Batch(
		func() tea.Msg { s, err := c.Source(); return sourceMsg{s, err} },
		m.poll(-1, 0),
		func() tea.Msg { p, err := c.Last(); return profMsg{p, err, true} },
	)
}

// poll long-polls past seq, after waiting back (the back-off after an error).
func (m Model) poll(seq int, back time.Duration) tea.Cmd {
	c := m.c
	return func() tea.Msg {
		time.Sleep(back)
		st, err := c.State(seq)
		return stateMsg{st, err}
	}
}

// send posts one command; a second while one is in flight is refused here,
// as the server would refuse it with 409.
func (m Model) send(cmd string) (Model, tea.Cmd) {
	if m.inflight != "" {
		m.status = fmt.Sprintf("%q is still running", m.inflight)
		return m, nil
	}
	m.inflight = cmd
	c := m.c
	return m, func() tea.Msg { d, err := c.Do(cmd); return doneMsg{cmd, d, err} }
}

// apply takes st if it is newer, moves the cursor to a new stop, and asks for
// the stack once per stop.
func (m Model) apply(st debug.State) (Model, tea.Cmd) {
	if st.Seq <= m.st.Seq && m.st.Run != "" {
		return m, nil
	}
	m.st = st
	if st.Run != "stopped" || st.Stop == nil {
		return m, nil
	}
	key := fmt.Sprintf("%s %d %d", st.Stop.Reason, st.Stop.Off, st.Stop.PC)
	if key == m.btFor {
		return m, nil
	}
	m.btFor, m.frame = key, 0
	if st.Stop.Line > 0 {
		m.cursor = st.Stop.Line
	}
	return m.send("bt")
}

func is404(err error) bool {
	var se *StatusError
	return errors.As(err, &se) && se.Code == 404
}

func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.WindowSizeMsg:
		m.w, m.h = msg.Width, msg.Height
	case sourceMsg:
		if is404(msg.err) {
			m.noDebug = true
		} else if msg.err != nil {
			m.status = "source: " + msg.err.Error()
		} else {
			m.src = msg.src
		}
	case stateMsg:
		if is404(msg.err) {
			m.noDebug = true
			return m, nil // no debug target: stop polling
		}
		if msg.err != nil {
			m.status = "custos: " + msg.err.Error()
			return m, m.poll(m.st.Seq, time.Second)
		}
		if m.status != "" && m.inflight == "" && m.st.Run == "" {
			m.status = "" // reconnected
		}
		var cmd tea.Cmd
		m, cmd = m.apply(msg.st)
		return m, tea.Batch(cmd, m.poll(m.st.Seq, 0))
	case doneMsg:
		m.inflight = ""
		switch {
		case msg.err != nil:
			m.status = msg.cmd + ": " + msg.err.Error()
			return m, nil
		case !msg.d.OK:
			m.status = msg.cmd + ": " + msg.d.Error
		default:
			m.status = ""
		}
		return m.apply(msg.d.State)
	case profMsg:
		m.capturing = false
		if msg.err != nil {
			if !(msg.last && is404(msg.err)) {
				m.status = "profile: " + msg.err.Error()
			}
			return m, nil
		}
		m.prof, m.status = &msg.c, ""
	case tea.KeyMsg:
		return m.key(msg)
	}
	return m, nil
}

// breakKey is how State.Breaks names line n of the app's file.
func (m Model) breakKey(n int) string {
	return fmt.Sprintf("%s:%d", filepath.Base(m.src.File), n)
}

func (m Model) key(k tea.KeyMsg) (tea.Model, tea.Cmd) {
	s := k.String()
	if m.prompt != "" {
		switch k.Type {
		case tea.KeyEsc:
			m.prompt, m.input = "", ""
		case tea.KeyEnter:
			line, prompt := m.input, m.prompt
			m.prompt, m.input = "", ""
			if prompt == ":" {
				return m.send(line)
			}
			ms, err := strconv.Atoi(line)
			if err != nil || ms <= 0 {
				m.status = "a window is a positive number of ms"
				return m, nil
			}
			m.ms = ms
			return m.capture()
		case tea.KeyBackspace:
			if m.input != "" {
				m.input = m.input[:len(m.input)-1]
			}
		case tea.KeyRunes:
			m.input += string(k.Runes)
		case tea.KeySpace:
			m.input += " "
		}
		return m, nil
	}
	switch s {
	case "q", "ctrl+c":
		return m, tea.Quit
	case "tab":
		m.tab = 1 - m.tab
		return m, nil
	}
	if m.tab == profileTab {
		if s == "r" {
			m.prompt, m.input = "window ms: ", strconv.Itoa(m.ms)
		}
		return m, nil
	}
	switch s {
	case "up", "k":
		m.cursor = max(1, m.cursor-1)
	case "down", "j":
		m.cursor = min(max(1, len(m.src.Lines)), m.cursor+1)
	case "b":
		verb := "break"
		if slices.Contains(m.st.Breaks, m.breakKey(m.cursor)) {
			verb = "clear"
		}
		return m.send(fmt.Sprintf("%s %d", verb, m.cursor))
	case "c":
		return m.send("continue")
	case "s":
		return m.send("step")
	case "n":
		return m.send("next")
	case "f":
		return m.send("finish")
	case "p":
		return m.send("pause")
	case "[", "]":
		if s == "[" {
			m.frame = max(0, m.frame-1)
		} else {
			m.frame = min(max(0, len(m.st.Frames)-1), m.frame+1)
		}
		return m.send(fmt.Sprintf("locals %d", m.frame))
	case ":":
		m.prompt = ":"
	}
	return m, nil
}

func (m Model) capture() (Model, tea.Cmd) {
	if m.capturing {
		m.status = "a capture is still running"
		return m, nil
	}
	m.capturing = true
	c, ms := m.c, m.ms
	return m, func() tea.Msg { p, err := c.Capture(ms); return profMsg{p, err, false} }
}
```

```go
package tui

import (
	"encoding/json"
	"net/http"
	"net/http/httptest"
	"strconv"
	"sync"
	"testing"
	"time"

	"github.com/davidg238/rete/custos/internal/custos"
	"github.com/davidg238/rete/custos/internal/debug"
)

type fake struct {
	mu   sync.Mutex
	st   debug.State
	cmds []string
	hold chan struct{} // when non-nil, /debug/do waits on it
}

func (f *fake) set(st debug.State) { f.mu.Lock(); f.st = st; f.mu.Unlock() }

func newFake(t *testing.T) (*fake, *httptest.Server) {
	f := &fake{st: debug.State{Seq: 1, Run: "boot"}}
	mux := http.NewServeMux()
	js := func(w http.ResponseWriter, v any) { json.NewEncoder(w).Encode(v) }
	mux.HandleFunc("GET /debug/source", func(w http.ResponseWriter, r *http.Request) {
```

```go
// Copyright (c) 2026 Ekorau LLC

package tui

// view.go — the screens. Colours are the terminal's 16 ANSI colours only, so
// an Omarchy theme (which sets the terminal palette) carries through (D3).

import (
	"fmt"
	"slices"
	"strings"

	"github.com/charmbracelet/lipgloss"
)

var (
	dim     = lipgloss.NewStyle().Foreground(lipgloss.Color("8"))
	accent  = lipgloss.NewStyle().Foreground(lipgloss.Color("4")).Bold(true)
	warn    = lipgloss.NewStyle().Foreground(lipgloss.Color("3"))
	bad     = lipgloss.NewStyle().Foreground(lipgloss.Color("1"))
	here    = lipgloss.NewStyle().Foreground(lipgloss.Color("2")).Bold(true)
	cursorS = lipgloss.NewStyle().Reverse(true)
	box     = lipgloss.NewStyle().Border(lipgloss.RoundedBorder()).BorderForeground(lipgloss.Color("8"))
)

// pane boxes body to w×h outer cells, titled; sizes clamp so a tiny terminal
// renders rather than panics.
func pane(title, body string, w, h int) string {
	w, h = max(w-2, 1), max(h-2, 1)
	lines := strings.Split(body, "\n")
	if len(lines) > h {
		lines = lines[len(lines)-h:]
	}
	for i, l := range lines {
		lines[i] = lipgloss.NewStyle().MaxWidth(w).Render(l)
	}
	return box.Width(w).Height(h).Render(accent.Render(title) + "\n" + strings.Join(lines, "\n"))
}

func (m Model) View() string {
	w, h := max(m.w, 20), max(m.h, 6)
	var tabs string
	if m.tab == debugTab {
		tabs = cursorS.Render(" debug ") + dim.Render(" profile ")
	} else {
		tabs = dim.Render(" debug ") + cursorS.Render(" profile ")
	}
	var body string
	if m.tab == debugTab {
		body = m.debugView(w, h-3)
	} else {
		body = m.profileView(w, h-3)
	}
	return lipgloss.JoinVertical(lipgloss.Left, tabs, body, m.statusLine(), m.help())
}

func (m Model) statusLine() string {
	if m.prompt != "" {
		return m.prompt + m.input + "█"
	}
	s := m.st.Run
	if m.st.Run == "stopped" && m.st.Stop != nil {
		s = fmt.Sprintf("stopped (%s) %s:%d  %s", m.st.Stop.Reason, m.st.Stop.File, m.st.Stop.Line, m.st.Stop.Method)
	}
	if m.noDebug && m.tab == debugTab {
		s = "no debug target (start custos with --debug-host or --debug-port)"
	}
	if m.inflight != "" {
		s += dim.Render("  … " + m.inflight)
	}
	if m.status != "" {
		s += "  " + bad.Render(m.status)
	}
	return s
}

func (m Model) help() string {
	if m.tab == profileTab {
		return dim.Render("r capture · tab debug · q quit")
	}
	return dim.Render("↑↓ line · b break · c continue · s step · n next · f finish · p pause · [ ] frame · : command · tab profile · q quit")
}

func (m Model) debugView(w, h int) string {
	tail := 6
	top := max(h-tail, 3)
	left := w * 3 / 5
	right := w - left
	src := pane("source", m.sourceBody(top-2), left, top)
	stack := pane("stack", m.stackBody(), right, top/2)
	locals := pane("locals", m.localsBody(), right, top-top/2)
	upper := lipgloss.JoinHorizontal(lipgloss.Top, src, lipgloss.JoinVertical(lipgloss.Left, stack, locals))
	return lipgloss.JoinVertical(lipgloss.Left, upper, pane("transcript", strings.Join(m.st.Transcript, "\n"), w, tail))
}

// sourceBody is the window of lines around the cursor, rows tall.
func (m Model) sourceBody(rows int) string {
	n := len(m.src.Lines)
	if n == 0 {
		return dim.Render("(no source)")
	}
	rows = max(rows, 1)
	first := min(max(1, m.cursor-rows/2), max(1, n-rows+1))
	var b strings.Builder
	for i := first; i <= n && i < first+rows; i++ {
		mark := " "
		if slices.Contains(m.st.Breaks, m.breakKey(i)) {
			mark = bad.Render("●")
		}
		if m.st.Stop != nil && m.st.Run == "stopped" && m.st.Stop.Line == i {
			mark = here.Render("▶")
		}
		num := fmt.Sprintf("%4d ", i)
		if !m.src.Lines[i-1].Code {
			num = dim.Render(num)
		}
		text := m.src.Lines[i-1].Text
		if i == m.cursor {
			text = cursorS.Render(text)
		}
		fmt.Fprintf(&b, "%s%s %s\n", mark, num, text)
	}
	return strings.TrimRight(b.String(), "\n")
}

func (m Model) stackBody() string {
	var b strings.Builder
	for i, f := range m.st.Frames {
		line := fmt.Sprintf("#%d %s", i, f.Where)
		if i == m.frame {
			line = cursorS.Render(line)
		}
		b.WriteString(line + "\n")
	}
	return strings.TrimRight(b.String(), "\n")
}

func (m Model) localsBody() string {
	if m.st.Locals == nil {
		return ""
	}
	var b strings.Builder
	b.WriteString(dim.Render(fmt.Sprintf("frame %d", m.st.Locals.Frame)) + "\n")
	for _, v := range m.st.Locals.Vars {
		fmt.Fprintf(&b, "%s = %s\n", v.Name, v.Value)
	}
	return strings.TrimRight(b.String(), "\n")
}

func (m Model) profileView(w, h int) string {
	if m.prof == nil {
		msg := "no capture yet — press r"
		if m.capturing {
			msg = "capturing…"
		}
		return pane("profile", dim.Render(msg), w, h)
	}
	r := m.prof.Report
	var b strings.Builder
	if r.Header != nil {
		fmt.Fprintf(&b, "%d ms  window=%s methods=%d activations=%d\n", m.prof.Ms, r.Header.Window, r.Header.Methods, r.Header.Activations)
	}
	for _, wn := range r.Warnings {
		b.WriteString(warn.Render(wn) + "\n")
	}
	for _, row := range r.Rows {
		fmt.Fprintf(&b, "%8d %5.1f%%  %s\n", row.Count, row.Pct, row.Name)
	}
	body := strings.TrimRight(b.String(), "\n")
	if lines := strings.Split(body, "\n"); len(lines) > h-2 { // keep the top of a ranking, not its tail
		body = strings.Join(lines[:max(h-2, 1)], "\n")
	}
	return pane("profile", body, w, h)
}
```

```go
// Run is `custos tui`. args excludes the verb; it answers the exit status.
func Run(args []string, stderr io.Writer) int {

		return 2
	}
	if _, err := tea.NewProgram(New(NewClient(*url)), tea.WithAltScreen()).Run(); err != nil {
		fmt.Fprintln(stderr, "custos tui:", err)
		return 1
	}
	return 0
}
```
